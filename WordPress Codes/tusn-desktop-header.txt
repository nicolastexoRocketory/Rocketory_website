// 1. HELPER: RELATIVE → ABSOLUTE URL
function make_absolute_url($path) {
    if (!$path) return '';
    $path = trim($path);
    if (preg_match('#^https?://#i', $path)) return $path;
    if (strpos($path, '/') === 0) return home_url($path);
    return get_stylesheet_directory_uri() . '/' . $path;
}

// 3. WALKER KLASSE (MIT STANDARD CLASSES & ACTIVE FIX)
class Custom_Header_Walker extends Walker_Nav_Menu {
    public function start_lvl(&$output, $depth = 0, $args = null) {
        $output .= '<ul class="sub-menu">';
    }
    public function start_el(&$output, $item, $depth = 0, $args = null, $id = 0) {
        $classes = empty($item->classes) ? array() : (array) $item->classes;
        
        // WP-FIX: Startseiten-Active-State erzwingen (falls es ein Custom Link ist)
        $is_home_link = (rtrim($item->url, '/') === rtrim(home_url(), '/') || $item->url === '/');
        if ( is_front_page() && $is_home_link ) {
            $classes[] = 'current-menu-item';
            $classes[] = 'current_page_item';
        }

        // Duplikate filtern
        $classes = array_unique($classes);
        $class_names = join(' ', apply_filters('nav_menu_css_class', array_filter($classes), $item, $args));
        $class_names = $class_names ? ' class="' . esc_attr($class_names) . '"' : '';

        $icon_url = get_post_meta($item->ID, '_submenu_icon_url', true);
        
        // Logik für Icon Positionierung
        $icon_left = '';
        $icon_right = '';

        if ($icon_url) {
            if ($depth > 0) {
                // Submenü: Icon Links
                $icon_left = '<img src="'.esc_url($icon_url).'" class="menu-icon-img">';
            } else {
                // Hauptmenü: Icon Rechts
                $icon_right = '<img src="'.esc_url($icon_url).'" class="menu-icon-right">';
            }
        }
        
        $output .= '<li' . $class_names . '>';
        $output .= '<a href="'.esc_url($item->url).'" title="'.esc_attr($item->title).'">';
        $output .= $icon_left . '<span class="link-text">' . esc_html($item->title) . '</span>' . $icon_right;
        $output .= '</a>';
    }
}
