function pt_register_team_cpt() {
    $tax_loc_labels = array( 'name' => 'Team Namen', 'singular_name' => 'Team Name', 'menu_name' => 'Team Namen' );
    $tax_loc_args = array( 'hierarchical' => true, 'labels' => $tax_loc_labels, 'show_ui' => true, 'show_admin_column' => true, 'rewrite' => array( 'slug' => 'team-name' ) );
    register_taxonomy( 'praxisteam_location', array( 'praxisteam_member' ), $tax_loc_args );

    $cpt_labels = array( 'name' => 'Handball Team', 'singular_name' => 'Teammitglied', 'menu_name' => 'Handball Team', 'add_new' => 'Neu hinzufügen', 'add_new_item' => 'Neues Teammitglied hinzufügen', 'all_items' => 'Alle Mitglieder' );
    $cpt_args = array( 'labels' => $cpt_labels, 'public' => true, 'show_ui' => true, 'show_in_menu' => true, 'rewrite' => array( 'slug' => 'teammitglied' ), 'has_archive' => false, 'hierarchical' => false, 'menu_position' => 20, 'menu_icon' => 'dashicons-groups', 'supports' => array( 'title', 'editor', 'thumbnail', 'page-attributes' ) );
    register_post_type( 'praxisteam_member', $cpt_args );
}
add_action( 'init', 'pt_register_team_cpt' );

function pt_save_player_details_meta_box( $post_id ) {
    if ( ! isset( $_POST['pt_player_details_nonce'] ) || ! wp_verify_nonce( $_POST['pt_player_details_nonce'], 'pt_save_player_details' ) ) {
        return;
    }
    if ( defined( 'DOING_AUTOSAVE' ) && DOING_AUTOSAVE ) {
        return;
    }
    if ( ! current_user_can( 'edit_post', $post_id ) ) {
        return;
    }

    $fields = array(
        'pt_player_number',
        'pt_player_position',
        'pt_player_birthday',
        'pt_player_nationality',
        'pt_player_height',
        'pt_sponsor_image'
    );

    foreach ( $fields as $field ) {
        if ( isset( $_POST[ $field ] ) ) {
            $val = wp_unslash( $_POST[ $field ] );
            if ( 'pt_sponsor_image' === $field ) {
                update_post_meta( $post_id, $field, esc_url_raw( $val ) );
            } else {
                update_post_meta( $post_id, $field, sanitize_text_field( $val ) );
            }
        }
    }
}
add_action( 'save_post', 'pt_save_player_details_meta_box' );

add_filter( 'manage_praxisteam_member_posts_columns', 'pt_set_custom_edit_team_columns' );
function pt_set_custom_edit_team_columns( $columns ) {
    $new = array();
    foreach ( $columns as $key => $label ) {
        $new[ $key ] = $label;
        if ( 'title' === $key ) {
            $new['pt_number']   = 'Nr.';
            $new['pt_position'] = 'Position';
            $new['menu_order']  = 'Reihenfolge';
        }
    }
    if ( ! isset( $new['menu_order'] ) ) {
        $new['menu_order'] = 'Reihenfolge';
    }
    return $new;
}
add_action( 'manage_praxisteam_member_posts_custom_column', 'pt_custom_team_column', 10, 2 );
function pt_custom_team_column( $column, $post_id ) {
    if ( 'menu_order' === $column ) {
        $post = get_post( $post_id );
        echo esc_html( $post->menu_order );
    }
    if ( 'pt_number' === $column ) {
        echo esc_html( get_post_meta( $post_id, 'pt_player_number', true ) );
    }
    if ( 'pt_position' === $column ) {
        echo esc_html( get_post_meta( $post_id, 'pt_player_position', true ) );
    }
}

function pt_save_team_member_order( $order_map ) {
	if ( ! is_array( $order_map ) || empty( $order_map ) ) {
		return;
	}

	foreach ( $order_map as $post_id => $menu_order ) {
		$post_id = intval( $post_id );
		if ( $post_id <= 0 ) {
			continue;
		}
		if ( 'praxisteam_member' !== get_post_type( $post_id ) ) {
			continue;
		}
		wp_update_post(
			array(
				'ID'         => $post_id,
				'menu_order' => intval( $menu_order ),
			)
		);
	}
}

function pt_save_team_location_order( $term_ids ) {
	if ( ! is_array( $term_ids ) ) {
		return;
	}

	$term_ids = array_values(
		array_filter(
			array_map( 'intval', $term_ids ),
			function ( $id ) {
				return $id > 0 && term_exists( $id, 'praxisteam_location' );
			}
		)
	);

	update_option( 'pt_team_location_order', $term_ids );
}

function pt_get_team_locations_ordered( $hide_empty = false ) {
	$terms = get_terms(
		array(
			'taxonomy'   => 'praxisteam_location',
			'hide_empty' => (bool) $hide_empty,
		)
	);

	if ( empty( $terms ) || is_wp_error( $terms ) ) {
		return array();
	}

	$saved_order = get_option( 'pt_team_location_order', array() );
	$saved_order = array_map( 'intval', (array) $saved_order );

	$terms_by_id = array();
	foreach ( $terms as $term ) {
		$terms_by_id[ $term->term_id ] = $term;
	}

	$ordered = array();
	foreach ( $saved_order as $term_id ) {
		if ( isset( $terms_by_id[ $term_id ] ) ) {
			$ordered[] = $terms_by_id[ $term_id ];
			unset( $terms_by_id[ $term_id ] );
		}
	}

	foreach ( $terms_by_id as $term ) {
		$ordered[] = $term;
	}

	if ( empty( $saved_order ) ) {
		usort(
			$ordered,
			function ( $a, $b ) {
				return strcasecmp( $a->name, $b->name );
			}
		);
	}

	return $ordered;
}

    <script>
        document.addEventListener('DOMContentLoaded', function() {
            var groupTitles = document.querySelectorAll('.pt-group-title');
            groupTitles.forEach(function(title) {
                title.addEventListener('click', function() {
                    this.parentElement.classList.toggle('is-open');
                });
            });

            var cards = document.querySelectorAll('.pt-card');
            var modal = document.getElementById('pt-modal');
            if (!modal) return;
            var overlay = modal.querySelector('.pt-modal-overlay');
            var closeBtn = modal.querySelector('.pt-modal-close');

            function openModal() {
                modal.classList.add('is-active');
                modal.setAttribute('aria-hidden', 'false');
                document.body.style.overflow = 'hidden';
            }

            function closeModal() {
                modal.classList.remove('is-active');
                modal.setAttribute('aria-hidden', 'true');
                document.body.style.overflow = '';
            }

            cards.forEach(function(card) {
                card.addEventListener('click', function() {
                    var name = this.getAttribute('data-name');
                    var img = this.getAttribute('data-img');
                    var sponsor = this.getAttribute('data-sponsor');
                    var statsMap = ['number', 'position', 'birthday', 'nationality', 'height'];

                    document.getElementById('pt-modal-name').textContent = name || '';
                    var imgEl = document.getElementById('pt-modal-img');
                    imgEl.src = img || '';
                    imgEl.alt = name || '';
                    imgEl.style.display = img ? 'block' : 'none';

                    statsMap.forEach(function(stat) {
                        var val = this.getAttribute('data-' + stat);
                        var row = document.getElementById('pt-modal-row-' + stat);
                        if (val && val.trim() !== '') {
                            document.getElementById('pt-modal-' + stat).textContent = val;
                            row.style.display = 'flex';
                        } else {
                            row.style.display = 'none';
                        }
                    }.bind(this));

                    var sponsorWrapper = document.getElementById('pt-modal-sponsor-wrapper');
                    if (sponsor && sponsor.trim() !== '') {
                        document.getElementById('pt-modal-sponsor-img').src = sponsor;
                        sponsorWrapper.style.display = 'block';
                    } else {
                        sponsorWrapper.style.display = 'none';
                    }

                    openModal();
                });
            });

            if (closeBtn) closeBtn.addEventListener('click', closeModal);
            if (overlay) overlay.addEventListener('click', closeModal);

            document.addEventListener('keydown', function(event) {
                if (event.key === 'Escape' && modal.classList.contains('is-active')) {
                    closeModal();
                }
            });
        });
    </script>
