define( 'TUSN_MAGAZIN_OPTION', 'tusn_magazin_settings' );

function tusn_magazin_defaults() {
	return array(
		'pdf_id'  => 0,
		'pdf_url' => '',
		'title'   => '',
		'history' => array(),
	);
}

function tusn_magazin_get_settings() {
	$saved = get_option( TUSN_MAGAZIN_OPTION, array() );
	if ( ! is_array( $saved ) ) {
		$saved = array();
	}

	$settings = wp_parse_args( $saved, tusn_magazin_defaults() );
	if ( ! is_array( $settings['history'] ) ) {
		$settings['history'] = array();
	}

	$settings['history'] = tusn_magazin_normalize_history( $settings['history'] );

	return $settings;
}

function tusn_magazin_normalize_history( $history ) {
	if ( ! is_array( $history ) ) {
		return array();
	}

	$normalized = array();

	foreach ( $history as $entry ) {
		if ( ! is_array( $entry ) ) {
			continue;
		}

		$pdf_id  = intval( $entry['pdf_id'] ?? 0 );
		$pdf_url = isset( $entry['pdf_url'] ) ? esc_url_raw( $entry['pdf_url'] ) : '';

		if ( $pdf_id > 0 ) {
			$attachment_url = wp_get_attachment_url( $pdf_id );
			if ( $attachment_url ) {
				$pdf_url = esc_url_raw( $attachment_url );
			} else {
				$pdf_id = 0;
			}
		}

		if ( ! $pdf_url ) {
			continue;
		}

		$key = isset( $entry['key'] ) ? sanitize_key( $entry['key'] ) : '';
		if ( ! $key ) {
			$key = 'hist_' . wp_generate_password( 8, false, false );
		}

		$normalized[] = array(
			'key'     => $key,
			'pdf_id'  => $pdf_id,
			'pdf_url' => $pdf_url,
			'title'   => isset( $entry['title'] ) ? sanitize_text_field( $entry['title'] ) : '',
			'date'    => isset( $entry['date'] ) ? intval( $entry['date'] ) : time(),
		);
	}

	usort(
		$normalized,
		function ( $a, $b ) {
			return intval( $b['date'] ) <=> intval( $a['date'] );
		}
	);

	return $normalized;
}

function tusn_magazin_get_pdf_url( $settings = null ) {
	if ( null === $settings ) {
		$settings = tusn_magazin_get_settings();
	}

	$pdf_id = intval( $settings['pdf_id'] ?? 0 );
	if ( $pdf_id > 0 ) {
		$url = wp_get_attachment_url( $pdf_id );
		if ( $url ) {
			return $url;
		}
	}

	$url = isset( $settings['pdf_url'] ) ? esc_url_raw( $settings['pdf_url'] ) : '';
	return $url ? $url : '';
}

function tusn_magazin_get_pdf_label( $pdf_id, $pdf_url, $title = '' ) {
	if ( $title ) {
		return $title;
	}

	if ( $pdf_id > 0 ) {
		$filename = basename( get_attached_file( $pdf_id ) );
		if ( $filename ) {
			return $filename;
		}
	}

	if ( $pdf_url ) {
		return basename( wp_parse_url( $pdf_url, PHP_URL_PATH ) );
	}

	return __( 'Magazin-Ausgabe', 'tusn-magazin' );
}

function tusn_magazin_history_has_pdf( $history, $pdf_id, $pdf_url ) {
	foreach ( $history as $entry ) {
		if ( $pdf_id > 0 && intval( $entry['pdf_id'] ?? 0 ) === $pdf_id ) {
			return true;
		}
		if ( $pdf_url && ! empty( $entry['pdf_url'] ) && $entry['pdf_url'] === $pdf_url ) {
			return true;
		}
	}
	return false;
}

function tusn_magazin_create_history_entry( $pdf_id, $pdf_url, $title = '' ) {
	return array(
		'key'     => 'hist_' . wp_generate_password( 8, false, false ),
		'pdf_id'  => intval( $pdf_id ),
		'pdf_url' => esc_url_raw( $pdf_url ),
		'title'   => sanitize_text_field( $title ),
		'date'    => time(),
	);
}

function tusn_magazin_get_editions( $settings = null ) {
	if ( null === $settings ) {
		$settings = tusn_magazin_get_settings();
	}

	$editions = array();
	$current_url = tusn_magazin_get_pdf_url( $settings );

	if ( $current_url ) {
		$editions[] = array(
			'key'   => 'current',
			'pdf'   => $current_url,
			'title' => tusn_magazin_get_pdf_label(
				intval( $settings['pdf_id'] ?? 0 ),
				$current_url,
				$settings['title'] ?? ''
			),
			'date'  => time(),
		);
	}

	foreach ( $settings['history'] as $entry ) {
		$editions[] = array(
			'key'   => $entry['key'],
			'pdf'   => $entry['pdf_url'],
			'title' => tusn_magazin_get_pdf_label(
				intval( $entry['pdf_id'] ?? 0 ),
				$entry['pdf_url'],
				$entry['title'] ?? ''
			),
			'date'  => intval( $entry['date'] ?? 0 ),
		);
	}

	return $editions;
}

add_action( 'admin_enqueue_scripts', 'tusn_magazin_admin_assets' );
function tusn_magazin_admin_assets( $hook ) {
	if ( 'toplevel_page_tusn-magazin' !== $hook ) {
		return;
	}

	wp_enqueue_media();
	wp_enqueue_script( 'jquery' );
	add_action( 'admin_footer', 'tusn_magazin_admin_js' );
}

function tusn_magazin_admin_js() {
	?>
	<script>
	jQuery(function ($) {
		var frame;

		function setPdfSelection(id, url, title) {
			$('#tusn_magazin_pdf_id').val(id || 0);
			$('#tusn_magazin_pdf_url').val(url || '');
			$('#tusn-magazin-pdf-name').text(title || '<?php echo esc_js( __( 'Keine PDF ausgewählt', 'tusn-magazin' ) ); ?>');
			$('#tusn-magazin-remove-pdf').toggle(!!url);
		}

		$('#tusn-magazin-select-pdf').on('click', function (e) {
			e.preventDefault();

			if (frame) {
				frame.open();
				return;
			}

			frame = wp.media({
				title: '<?php echo esc_js( __( 'PDF für Magazin auswählen', 'tusn-magazin' ) ); ?>',
				button: { text: '<?php echo esc_js( __( 'PDF verwenden', 'tusn-magazin' ) ); ?>' },
				library: { type: 'application/pdf' },
				multiple: false
			});

			frame.on('select', function () {
				var attachment = frame.state().get('selection').first().toJSON();
				setPdfSelection(attachment.id, attachment.url, attachment.filename || attachment.title);
			});

			frame.open();
		});

		$('#tusn-magazin-remove-pdf').on('click', function (e) {
			e.preventDefault();
			setPdfSelection(0, '', '');
		});
	});
	</script>
	<?php
}

function tusn_magazin_admin_page() {
	if ( ! current_user_can( 'manage_options' ) ) {
		return;
	}

	$settings = tusn_magazin_get_settings();

	if ( isset( $_GET['tusn_magazin_delete'] ) && isset( $_GET['key'] ) ) {
		$delete_key = sanitize_key( wp_unslash( $_GET['key'] ) );
		check_admin_referer( 'tusn_magazin_delete_' . $delete_key );

		$settings['history'] = array_values(
			array_filter(
				$settings['history'],
				function ( $entry ) use ( $delete_key ) {
					return ( $entry['key'] ?? '' ) !== $delete_key;
				}
			)
		);

		update_option( TUSN_MAGAZIN_OPTION, $settings );
		echo '<div class="notice notice-success is-dismissible"><p>' . esc_html__( 'Ausgabe aus dem Archiv entfernt.', 'tusn-magazin' ) . '</p></div>';
	}

	if ( isset( $_POST['tusn_magazin_save'] ) && check_admin_referer( 'tusn_magazin_save', 'tusn_magazin_nonce' ) ) {
		$old_settings = $settings;
		$old_pdf_id   = intval( $old_settings['pdf_id'] ?? 0 );
		$old_pdf_url  = tusn_magazin_get_pdf_url( $old_settings );
		$history      = $old_settings['history'];

		$pdf_id  = isset( $_POST['pdf_id'] ) ? intval( $_POST['pdf_id'] ) : 0;
		$pdf_url = '';

		if ( $pdf_id > 0 ) {
			$attachment_url = wp_get_attachment_url( $pdf_id );
			if ( $attachment_url ) {
				$pdf_url = esc_url_raw( $attachment_url );
			} else {
				$pdf_id = 0;
			}
		}

		$new_title = isset( $_POST['title'] ) ? sanitize_text_field( wp_unslash( $_POST['title'] ) ) : '';

		if ( $pdf_id > 0 && $old_pdf_id > 0 && $pdf_id !== $old_pdf_id && $old_pdf_url ) {
			if ( ! tusn_magazin_history_has_pdf( $history, $old_pdf_id, $old_pdf_url ) ) {
				$history[] = tusn_magazin_create_history_entry(
					$old_pdf_id,
					$old_pdf_url,
					$old_settings['title'] ?? ''
				);
			}
		}

		if ( $pdf_id > 0 && $pdf_url ) {
			$history = array_values(
				array_filter(
					$history,
					function ( $entry ) use ( $pdf_id, $pdf_url ) {
						$entry_id  = intval( $entry['pdf_id'] ?? 0 );
						$entry_url = $entry['pdf_url'] ?? '';
						if ( $entry_id > 0 && $entry_id === $pdf_id ) {
							return false;
						}
						if ( $entry_url && $entry_url === $pdf_url ) {
							return false;
						}
						return true;
					}
				)
			);
		}

		$settings = array(
			'pdf_id'  => $pdf_id,
			'pdf_url' => $pdf_url,
			'title'   => $new_title,
			'history' => tusn_magazin_normalize_history( $history ),
		);

		update_option( TUSN_MAGAZIN_OPTION, $settings );
		echo '<div class="notice notice-success is-dismissible"><p>' . esc_html__( 'Magazin gespeichert.', 'tusn-magazin' ) . '</p></div>';
	}

	$pdf_url      = tusn_magazin_get_pdf_url( $settings );
	$pdf_filename = tusn_magazin_get_pdf_label(
		intval( $settings['pdf_id'] ?? 0 ),
		$pdf_url,
		''
	);
	?>
	<div class="wrap tusn-magazin-admin">
		<h1><?php esc_html_e( 'Magazin', 'tusn-magazin' ); ?></h1>
		<p><?php esc_html_e( 'Laden Sie hier die aktuelle Magazin-PDF hoch. Beim Austausch wandert die bisherige Ausgabe automatisch ins Archiv. Im Frontend werden alle Ausgaben als Cover-Vorschau angezeigt.', 'tusn-magazin' ); ?></p>

		<form method="post">
			<?php wp_nonce_field( 'tusn_magazin_save', 'tusn_magazin_nonce' ); ?>

			<table class="form-table" role="presentation">
				<tr>
					<th scope="row"><label for="tusn_magazin_title"><?php esc_html_e( 'Titel (optional)', 'tusn-magazin' ); ?></label></th>
					<td>
						<input type="text" class="regular-text" id="tusn_magazin_title" name="title" value="<?php echo esc_attr( $settings['title'] ); ?>" placeholder="<?php esc_attr_e( 'z. B. Vereinsmagazin 2026', 'tusn-magazin' ); ?>">
						<p class="description"><?php esc_html_e( 'Wird oberhalb des Magazins und in der Cover-Vorschau angezeigt.', 'tusn-magazin' ); ?></p>
					</td>
				</tr>
				<tr>
					<th scope="row"><?php esc_html_e( 'Aktuelle PDF', 'tusn-magazin' ); ?></th>
					<td>
						<input type="hidden" id="tusn_magazin_pdf_id" name="pdf_id" value="<?php echo esc_attr( intval( $settings['pdf_id'] ) ); ?>">
						<input type="hidden" id="tusn_magazin_pdf_url" name="pdf_url" value="<?php echo esc_attr( $pdf_url ); ?>">
						<p><strong id="tusn-magazin-pdf-name"><?php echo esc_html( $pdf_url ? $pdf_filename : __( 'Keine PDF ausgewählt', 'tusn-magazin' ) ); ?></strong></p>
						<p>
							<button type="button" class="button button-secondary" id="tusn-magazin-select-pdf"><?php esc_html_e( 'PDF auswählen / hochladen', 'tusn-magazin' ); ?></button>
							<button type="button" class="button-link-delete" id="tusn-magazin-remove-pdf" <?php echo $pdf_url ? '' : 'style="display:none;"'; ?>><?php esc_html_e( 'PDF entfernen', 'tusn-magazin' ); ?></button>
						</p>
						<?php if ( $pdf_url ) : ?>
							<p><a href="<?php echo esc_url( $pdf_url ); ?>" target="_blank" rel="noopener"><?php esc_html_e( 'PDF im Browser öffnen', 'tusn-magazin' ); ?></a></p>
						<?php endif; ?>
					</td>
				</tr>
			</table>

			<?php submit_button( __( 'Speichern', 'tusn-magazin' ), 'primary', 'tusn_magazin_save' ); ?>
		</form>

		<?php if ( ! empty( $settings['history'] ) ) : ?>
			<hr>
			<h2><?php esc_html_e( 'Archiv – frühere Ausgaben', 'tusn-magazin' ); ?></h2>
			<p><?php esc_html_e( 'Diese PDFs wurden durch eine neue Ausgabe ersetzt und erscheinen im Frontend als weitere Cover.', 'tusn-magazin' ); ?></p>

			<table class="widefat striped tusn-magazin-history-table">
				<thead>
					<tr>
						<th scope="col"><?php esc_html_e( 'Titel / Datei', 'tusn-magazin' ); ?></th>
						<th scope="col"><?php esc_html_e( 'Archiviert am', 'tusn-magazin' ); ?></th>
						<th scope="col"><?php esc_html_e( 'PDF', 'tusn-magazin' ); ?></th>
						<th scope="col" style="width:120px;"><?php esc_html_e( 'Aktion', 'tusn-magazin' ); ?></th>
					</tr>
				</thead>
				<tbody>
					<?php foreach ( $settings['history'] as $entry ) : ?>
						<?php
						$entry_label = tusn_magazin_get_pdf_label(
							intval( $entry['pdf_id'] ?? 0 ),
							$entry['pdf_url'] ?? '',
							$entry['title'] ?? ''
						);
						$delete_url  = wp_nonce_url(
							add_query_arg(
								array(
									'page'                 => 'tusn-magazin',
									'tusn_magazin_delete'  => '1',
									'key'                  => $entry['key'],
								),
								admin_url( 'admin.php' )
							),
							'tusn_magazin_delete_' . $entry['key']
						);
						?>
						<tr>
							<td><strong><?php echo esc_html( $entry_label ); ?></strong></td>
							<td><?php echo esc_html( wp_date( get_option( 'date_format' ), intval( $entry['date'] ?? 0 ) ) ); ?></td>
							<td><a href="<?php echo esc_url( $entry['pdf_url'] ); ?>" target="_blank" rel="noopener"><?php esc_html_e( 'Öffnen', 'tusn-magazin' ); ?></a></td>
							<td>
								<a href="<?php echo esc_url( $delete_url ); ?>" class="submitdelete" onclick="return confirm('<?php echo esc_js( __( 'Diese Ausgabe aus dem Archiv entfernen?', 'tusn-magazin' ) ); ?>');">
									<?php esc_html_e( 'Löschen', 'tusn-magazin' ); ?>
								</a>
							</td>
						</tr>
					<?php endforeach; ?>
				</tbody>
			</table>
		<?php endif; ?>

		<hr>
		<h2><?php esc_html_e( 'Shortcode', 'tusn-magazin' ); ?></h2>
		<p><?php esc_html_e( 'Fügen Sie den folgenden Shortcode in eine Seite oder einen Beitrag ein (z. B. in Elementor):', 'tusn-magazin' ); ?></p>
		<input type="text" readonly value="[magazin]" class="large-text code" onclick="this.select();" style="max-width:320px;">
	</div>
	<?php
}

add_shortcode( 'magazin', 'tusn_magazin_shortcode' );
function tusn_magazin_shortcode( $atts ) {
	$atts = shortcode_atts(
		array(
			'pdf' => '',
		),
		$atts,
		'magazin'
	);

	$settings       = tusn_magazin_get_settings();
	$pdf_url        = $atts['pdf'] ? esc_url_raw( $atts['pdf'] ) : tusn_magazin_get_pdf_url( $settings );
	$title          = $settings['title'] ?? '';
	$active_key     = 'current';
	$past_editions  = array();

	foreach ( $settings['history'] as $entry ) {
		$past_editions[] = array(
			'key'   => $entry['key'],
			'pdf'   => $entry['pdf_url'],
			'title' => tusn_magazin_get_pdf_label(
				intval( $entry['pdf_id'] ?? 0 ),
				$entry['pdf_url'] ?? '',
				$entry['title'] ?? ''
			),
			'date'  => intval( $entry['date'] ?? 0 ),
		);
	}

		$past_editions = array();

			? '<p><em>' . esc_html__( 'Magazin: Bitte im Backend unter „Magazin“ eine PDF hochladen.', 'tusn-magazin' ) . '</em></p>'
			: '';
	}

	tusn_magazin_enqueue_assets();

	$instance_id = 'tusn-magazin-' . wp_unique_id();

	ob_start();
	?>
	<div
		class="tusn-magazin"
		id="<?php echo esc_attr( $instance_id ); ?>"
		data-pdf="<?php echo esc_url( $pdf_url ); ?>"
		data-active="<?php echo esc_attr( $active_key ); ?>"

						<div class="tusn-magazin__book"></div>
					</div>
				</div>

			<p class="tusn-magazin__error" hidden></p>

function tusn_magazin_frontend_js() {
	$load_error = esc_js( __( 'Das Magazin konnte nicht geladen werden.', 'tusn-magazin' ) );

	return <<<JS
(function () {
	var coverCache = {};
	var pdfDocCache = {};
	var INITIAL_PAGES = 5;
	var ZOOM_MIN = 1;
	var ZOOM_MAX = 2.5;
	var ZOOM_STEP = 0.25;

	function clamp(value, min, max) {
		return Math.min(max, Math.max(min, value));
	}

	function roundZoom(value) {
		return Math.round(value * 100) / 100;
	}

	function initMagazin(root) {
		if (!root || root.dataset.magazinBooted === '1') {
			return;
		}
		root.dataset.magazinBooted = '1';

		var loader = root.querySelector('.tusn-magazin__loader');
		var bookWrap = root.querySelector('.tusn-magazin__book-wrap');
		var bookEl = root.querySelector('.tusn-magazin__book');
		var stage = root.querySelector('.tusn-magazin__stage');
		var zoomViewport = root.querySelector('.tusn-magazin__zoom-viewport');
		var zoomLayer = root.querySelector('.tusn-magazin__zoom-layer');
		var errorEl = root.querySelector('.tusn-magazin__error');
		var controls = root.querySelector('.tusn-magazin__controls');
		var btnPrev = root.querySelector('.tusn-magazin__btn--prev');

		var btnDownload = root.querySelector('.tusn-magazin__btn--download');
		var zoomLabel = root.querySelector('.tusn-magazin__zoom-label');
		var pageCurrent = root.querySelector('.tusn-magazin__page-current');
		var pageTotal = root.querySelector('.tusn-magazin__page-total');
		var archiveItems = root.querySelectorAll('.tusn-magazin__archive-item');
		var backCurrentBtn = root.querySelector('.tusn-magazin__back-current');
		var preloadHint = root.querySelector('.tusn-magazin__preload-hint');
		var pageFlip = null;
		var mobileQuery = window.matchMedia('(max-width: 768px)');
		var currentPdfUrl = root.getAttribute('data-pdf') || '';
		var activePdf = currentPdfUrl;
		var isLoading = false;
		var isLoadingMore = false;
		var backgroundToken = 0;
		var editionState = null;
		var zoom = 1;
		var panX = 0;
		var panY = 0;
		var isPanning = false;
		var panStartX = 0;
		var panStartY = 0;
		var panOriginX = 0;
		var panOriginY = 0;
		var syncFlipQueued = false;
		var pendingSyncIndex = 0;

		function isMagazinMobile() {
			return mobileQuery.matches;
		}

		function getRenderPixelRatio() {
			var dpr = window.devicePixelRatio || 1;
			if (isMagazinMobile()) {
				return Math.min(Math.max(dpr, 2), 3);
			}
			return Math.min(Math.max(dpr, 1.5), 2.5);
		}

		function getAvailableBookWidth(isMobile) {
			var measured = 0;
			var stageWidth = stage && stage.clientWidth ? stage.clientWidth : 0;
			var wrapWidth = bookWrap && bookWrap.clientWidth ? bookWrap.clientWidth : 0;
			var viewportWidth = zoomViewport && zoomViewport.clientWidth ? zoomViewport.clientWidth : 0;

			if (viewportWidth > 40) {
				measured = viewportWidth - (isMobile ? 8 : 16);
			} else if (wrapWidth > 40) {
				measured = wrapWidth - 8;
			} else if (stageWidth) {
				measured = stageWidth - (isMobile ? 28 : 32);
			} else {
				measured = window.innerWidth - (isMobile ? 72 : 80);
			}

			/* Desktop: Seitenbreite ≈ Hälfte der Stage → PageFlip zeigt Doppelseiten */
			if (!isMobile) {
				measured = Math.floor(measured / 2);
				return Math.max(240, Math.min(520, measured));
			}

			var cap = Math.max(280, window.innerWidth - 32);
			if (measured < 200) {
				measured = Math.max(200, window.innerWidth - 32);
			}
			return Math.max(200, Math.min(cap, Math.floor(measured)));
		}

		function prepareBookContainer() {
			if (!bookWrap) return;
			bookWrap.hidden = false;
			bookWrap.style.display = 'flex';
			bookWrap.style.visibility = 'visible';
			if (stage) {
				void stage.offsetWidth;
			}
			if (zoomViewport) {
				void zoomViewport.offsetWidth;
			}
			void bookWrap.offsetWidth;
		}

		function refreshFlipbookLayout() {
			if (!pageFlip) return;
			if (stage) {
				void stage.offsetWidth;
			}
			if (bookWrap) {
				void bookWrap.offsetWidth;
			}
			if (typeof pageFlip.update === 'function') {
				try {
					pageFlip.update();
				} catch (e) {}
			}
		}

		function scheduleFlipbookLayoutRefresh() {
			window.requestAnimationFrame(function () {
				refreshFlipbookLayout();
				window.requestAnimationFrame(function () {
					refreshFlipbookLayout();
					window.setTimeout(refreshFlipbookLayout, 120);
					window.setTimeout(refreshFlipbookLayout, 420);
				});
			});
		}

		function getMobileMinWidth(pageWidth) {
			return Math.max(200, Math.min(pageWidth || 520, getAvailableBookWidth(true)));
		}

		function openPdfInNewWindow() {
			if (!activePdf) return;
			window.open(activePdf, '_blank', 'noopener,noreferrer');

			root.classList.toggle('is-zoomed', zoom > 1.01);
			if (zoomLabel) {
				zoomLabel.textContent = Math.round(zoom * 100) + '%';
			}
			if (btnZoomOut) btnZoomOut.disabled = zoom <= ZOOM_MIN + 0.001;
			if (btnZoomIn) btnZoomIn.disabled = zoom >= ZOOM_MAX - 0.001;
		}

		function setZoom(nextZoom, keepPan) {
			zoom = roundZoom(clamp(nextZoom, ZOOM_MIN, ZOOM_MAX));
			if (!keepPan || zoom <= 1.01) {
				panX = 0;
				panY = 0;
			}
			applyZoomTransform();
		}

		function resetZoom() {
			setZoom(1, false);
		}

		function setPreloadHint(show) {
			isLoadingMore = !!show;
			root.classList.toggle('is-loading-more', !!show);
			if (preloadHint) {
				preloadHint.hidden = !show;
			}
		}

		function showLoader() {
			root.classList.remove('is-ready');
			root.classList.add('is-switching');
			if (loader) {
				loader.hidden = false;
				loader.style.display = '';
			}
			if (bookWrap) {
				bookWrap.hidden = true;
			}
			if (controls) {
				controls.hidden = true;
			}
			if (errorEl) {
				errorEl.hidden = true;
				errorEl.textContent = '';
			}
		}

		function hideLoader() {
			root.classList.remove('is-switching');
			root.classList.add('is-ready');
			if (loader) {
				loader.hidden = true;
				loader.style.display = 'none';
			}
		}

		function showError(message) {
			hideLoader();
			if (bookWrap) bookWrap.hidden = true;
			if (controls) controls.hidden = true;
			if (errorEl) {
				errorEl.textContent = message;
				errorEl.hidden = false;
			}
		}

		function destroyFlipbook() {
			if (pageFlip && typeof pageFlip.destroy === 'function') {
				try {
					pageFlip.destroy();
				} catch (e) {}
			}
			pageFlip = null;
			if (bookEl) {
				bookEl.innerHTML = '';
				bookEl.className = 'tusn-magazin__book';
				bookEl.removeAttribute('style');
			}
		}

		function isFlipAnimating() {
			if (!pageFlip || typeof pageFlip.getState !== 'function') {
				return false;
			}
			var state = pageFlip.getState();
			return state === 'flipping' || state === 'user_fold' || state === 'fold_corner';
		}

		function whenFlipIdle(callback) {
			var attempts = 0;
			function tick() {
				attempts++;
				if (!pageFlip || !isFlipAnimating() || attempts > 150) {
					callback();
					return;
				}
				window.requestAnimationFrame(tick);
			}
			tick();
		}

		function rebuildBookPageElements(images) {
			if (!bookEl) {
				return [];
			}
			bookEl.innerHTML = '';
			images.forEach(function (src) {
				var page = document.createElement('div');
				page.className = 'page';
				var img = document.createElement('img');
				img.src = src;
				img.alt = '';
				img.decoding = 'async';
				page.appendChild(img);
				bookEl.appendChild(page);
			});
			return bookEl.querySelectorAll('.page');
		}

		function attachFlipbookEvents() {
			if (!pageFlip) {
				return;
			}
			pageFlip.on('flip', function () {
				updateControls();
			});
			pageFlip.on('changeState', function (e) {
				if (e.data === 'read') {
					window.requestAnimationFrame(function () {
						refreshFlipbookLayout();
						updateControls();
					});
				}
			});
		}

		function tryFlipNext() {
			if (!pageFlip) {
				return;
			}
			var index = pageFlip.getCurrentPageIndex();
			var loadedCount = pageFlip.getPageCount();
			if (
				index >= loadedCount - 1 &&
				editionState &&
				editionState.loadedThrough > loadedCount
			) {
				queueSyncFlipbookPages(index + 1, function () {
					if (pageFlip && pageFlip.getCurrentPageIndex() < pageFlip.getPageCount() - 1) {
						pageFlip.flipNext();
					}
				});
				return;
			}
			pageFlip.flipNext();
		}

		function queueSyncFlipbookPages(preserveIndex, done) {
			pendingSyncIndex = typeof preserveIndex === 'number'
				? preserveIndex
				: (pageFlip ? pageFlip.getCurrentPageIndex() : 0);
			if (syncFlipQueued) {
				return;
			}
			syncFlipQueued = true;
			whenFlipIdle(function () {
				syncFlipQueued = false;
				syncFlipbookPages(pendingSyncIndex, function () {
					if (
						pageFlip &&
						editionState &&
						getLoadedImagesSlice().length > pageFlip.getPageCount()
					) {
						queueSyncFlipbookPages(pageFlip.getCurrentPageIndex(), done);
						return;
					}
					if (done) done();
				});
			});
		}

		function syncFlipbookPages(preserveIndex, done) {
			if (!editionState) {
				if (done) done();
				return;
			}
			var images = getLoadedImagesSlice();
			if (!images.length) {
				if (done) done();
				return;
			}

			var targetIndex = typeof preserveIndex === 'number'
				? preserveIndex
				: (pageFlip ? pageFlip.getCurrentPageIndex() : 0);

			if (!pageFlip) {
				mountFlipbook({
					images: images,
					width: editionState.width,
					height: editionState.height,
					isMobile: editionState.isMobile,
					totalPages: editionState.totalPages,
					preserveIndex: targetIndex
				});
				if (done) done();
				return;
			}

			var currentCount = pageFlip.getPageCount();
			if (images.length <= currentCount) {
				updateControls();
				if (done) done();
				return;
			}

			prepareBookContainer();
			var pages = rebuildBookPageElements(images);
			pageFlip.updateFromHtml(pages);
			pageFlip.turnToPage(Math.min(targetIndex, pages.length - 1));
			refreshFlipbookLayout();
			scheduleFlipbookLayoutRefresh();
			updateControls();
			if (done) done();
		}

		function updateControls() {
			if (!pageFlip) return;
			var index = pageFlip.getCurrentPageIndex();
			var loadedCount = pageFlip.getPageCount();
			var total = editionState && editionState.totalPages ? editionState.totalPages : loadedCount;
			var hasMoreLoaded = editionState && editionState.loadedThrough > loadedCount;
			if (pageCurrent) pageCurrent.textContent = String(index + 1);
			if (pageTotal) pageTotal.textContent = String(total);
			if (btnPrev) btnPrev.disabled = index <= 0;
			if (btnNext) {
				btnNext.disabled = index >= total - 1 || (index >= loadedCount - 1 && !hasMoreLoaded);
			}
		}

		function bindControls() {
			function blurButton(event) {
				if (event.currentTarget) {
					event.currentTarget.blur();
				}
			}

			if (btnPrev) {
				btnPrev.onclick = function (event) {
					if (pageFlip) pageFlip.flipPrev();
					blurButton(event);
				};
			}
			if (btnNext) {
				btnNext.onclick = function (event) {
					tryFlipNext();
					blurButton(event);
				};
			}
			if (btnZoomIn) {
				btnZoomIn.onclick = function (event) {
					setZoom(zoom + ZOOM_STEP, true);
					blurButton(event);
				};
			}
			if (btnZoomOut) {
				btnZoomOut.onclick = function (event) {
					setZoom(zoom - ZOOM_STEP, true);
					blurButton(event);
				};
			}
			if (btnDownload) {
				btnDownload.onclick = function (event) {
					openPdfInNewWindow();
					blurButton(event);
				};
			}
		}

		function setActiveEdition(key) {
			archiveItems.forEach(function (item) {
				var isActive = key !== 'current' && item.getAttribute('data-edition-key') === key;
				item.classList.toggle('is-active', isActive);
				item.setAttribute('aria-pressed', isActive ? 'true' : 'false');
			});
			root.setAttribute('data-active', key);
			if (backCurrentBtn) {
				backCurrentBtn.hidden = key === 'current';
			}
		}

		async function renderPdfCover(url, maxWidth) {
			if (coverCache[url]) {
				return coverCache[url];
			}

			if (!window.pdfjsLib) {
				throw new Error('PDF.js missing');
			}

			pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
			var loadingTask = pdfjsLib.getDocument({ url: url, withCredentials: false });
			var pdf = await loadingTask.promise;
			var page = await pdf.getPage(1);
			var viewport = page.getViewport({ scale: 1 });
			var scale = maxWidth / viewport.width;
			var scaledViewport = page.getViewport({ scale: scale });
			var canvas = document.createElement('canvas');
			var context = canvas.getContext('2d');

			canvas.width = Math.floor(scaledViewport.width);
			canvas.height = Math.floor(scaledViewport.height);

			await page.render({
				canvasContext: context,
				viewport: scaledViewport
			}).promise;

			var dataUrl = canvas.toDataURL('image/jpeg', 0.88);
			coverCache[url] = dataUrl;
			return dataUrl;
		}

		function loadArchiveCovers() {
			archiveItems.forEach(function (item) {
				var pdfUrl = item.getAttribute('data-edition-pdf');
				var coverWrap = item.querySelector('.tusn-magazin__archive-cover');
				if (!pdfUrl || !coverWrap || coverWrap.querySelector('img')) {
					return;
				}

				renderPdfCover(pdfUrl, 280)
					.then(function (src) {
						var img = document.createElement('img');
						img.src = src;
						img.alt = '';
						coverWrap.appendChild(img);
						item.classList.add('is-cover-ready');
					})
					.catch(function () {
						item.classList.add('is-cover-ready');
					});
			});
		}

		async function getPdfDocument(url) {
			if (pdfDocCache[url]) {
				return pdfDocCache[url];
			}

			if (!window.pdfjsLib) {
				throw new Error('PDF.js missing');
			}

			pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
			var loadingTask = pdfjsLib.getDocument({ url: url, withCredentials: false });
			var pdf = await loadingTask.promise;
			pdfDocCache[url] = pdf;
			return pdf;
		}

		async function renderPagesFromPdf(pdf, startPage, endPage, isMobile) {
			var images = [];
			var displayWidth = getAvailableBookWidth(isMobile);
			var pixelRatio = getRenderPixelRatio();
			var pageWidth = editionState ? editionState.width : 0;
			var pageHeight = editionState ? editionState.height : 0;

			for (var pageNum = startPage; pageNum <= endPage; pageNum++) {
				var page = await pdf.getPage(pageNum);
				var baseViewport = page.getViewport({ scale: 1 });
				var renderScale = (displayWidth * pixelRatio) / baseViewport.width;
				var scaledViewport = page.getViewport({ scale: renderScale });
				var canvas = document.createElement('canvas');
				var context = canvas.getContext('2d');

				canvas.width = Math.floor(scaledViewport.width);
				canvas.height = Math.floor(scaledViewport.height);

				if (!pageWidth) {
					pageWidth = Math.floor(displayWidth);
					pageHeight = Math.floor(displayWidth * (baseViewport.height / baseViewport.width));
				}

				await page.render({
					canvasContext: context,
					viewport: scaledViewport
				}).promise;

				images.push(canvas.toDataURL('image/jpeg', 0.94));
			}

			return {
				images: images,
				width: pageWidth,
				height: pageHeight
			};
		}

		function getLoadedImagesSlice() {
			if (!editionState || !editionState.loadedThrough) {
				return [];
			}
			return editionState.images.slice(0, editionState.loadedThrough);
		}

		function mountFlipbook(data, silent) {
			var PageFlipCtor = (window.St && window.St.PageFlip) ? window.St.PageFlip : window.PageFlip;
			if (!PageFlipCtor) {
				throw new Error('PageFlip missing');
			}

			var preserveIndex = typeof data.preserveIndex === 'number' ? data.preserveIndex : 0;
			destroyFlipbook();
			if (!silent) {
				resetZoom();
			}

			prepareBookContainer();

			var pages = rebuildBookPageElements(data.images);

			var mobileMaxHeight = Math.min(560, Math.max(320, window.innerHeight - 220));

			pageFlip = new PageFlipCtor(bookEl, {
				width: data.width,
				height: data.height,
				size: data.isMobile ? 'fixed' : 'stretch',
				minWidth: data.isMobile ? getMobileMinWidth(data.width) : 280,
				maxWidth: data.isMobile ? data.width : 1400,
				minHeight: data.isMobile ? Math.min(mobileMaxHeight, Math.max(280, data.height)) : 360,
				maxHeight: data.isMobile ? mobileMaxHeight : 1400,
				maxShadowOpacity: data.isMobile ? 0.25 : 0.5,
				showCover: !data.isMobile,
				mobileScrollSupport: true,
				usePortrait: !!data.isMobile,
				autoSize: true,
				drawShadow: !data.isMobile,
				flippingTime: data.isMobile ? 520 : 700
			});

			pageFlip.loadFromHTML(pages);
			attachFlipbookEvents();

			refreshFlipbookLayout();
			scheduleFlipbookLayoutRefresh();

			if (preserveIndex > 0) {
				pageFlip.turnToPage(Math.min(preserveIndex, pageFlip.getPageCount() - 1));
			}

			if (!silent) {
				hideLoader();
			}

			bookWrap.hidden = false;
			bookWrap.style.display = 'flex';
			controls.hidden = false;
			updateControls();
			if (!silent) {
				bindControls();
			}
			applyZoomTransform();
			root.setAttribute('tabindex', '0');
			scheduleFlipbookLayoutRefresh();
		}

		function remountFlipbook(preserveIndex) {
			queueSyncFlipbookPages(preserveIndex);
		}

		async function loadRemainingPages(pdfUrl, startPage, totalPages, token) {
			try {
				var pdf = editionState && editionState.pdf;
				if (!pdf || token !== backgroundToken) return;

				for (var start = startPage; start <= totalPages; start += INITIAL_PAGES) {
					if (token !== backgroundToken) return;
					var end = Math.min(start + INITIAL_PAGES - 1, totalPages);
					var batch = await renderPagesFromPdf(pdf, start, end, editionState.isMobile);
					if (token !== backgroundToken) return;

					for (var j = 0; j < batch.images.length; j++) {
						editionState.images[start - 1 + j] = batch.images[j];
					}
					editionState.loadedThrough = end;
					queueSyncFlipbookPages(pageFlip ? pageFlip.getCurrentPageIndex() : 0);
				}

				if (token !== backgroundToken) return;
				setPreloadHint(false);
				updateControls();
			} catch (e) {
				setPreloadHint(false);
			}
		}

		async function loadEdition(pdfUrl, editionKey, forceReload) {
			if (!pdfUrl || isLoading) {
				return;
			}
			if (!forceReload && pdfUrl === activePdf && pageFlip) {
				return;
			}

			isLoading = true;
			backgroundToken++;
			var token = backgroundToken;
			activePdf = pdfUrl;
			editionState = null;
			setActiveEdition(editionKey || 'current');
			setPreloadHint(false);
			showLoader();

			try {
				var isMobile = isMagazinMobile();
				root.classList.toggle('tusn-magazin--mobile', isMobile);
				var pdf = await getPdfDocument(pdfUrl);
				if (token !== backgroundToken) return;

				var totalPages = pdf.numPages;
				var firstEnd = Math.min(INITIAL_PAGES, totalPages);

				editionState = {
					pdfUrl: pdfUrl,
					pdf: pdf,
					totalPages: totalPages,
					images: new Array(totalPages),
					width: 0,
					height: 0,
					isMobile: isMobile,
					loadedThrough: 0
				};

				var batch = await renderPagesFromPdf(pdf, 1, firstEnd, isMobile);
				if (token !== backgroundToken) return;

				editionState.width = batch.width;
				editionState.height = batch.height;
				for (var i = 0; i < batch.images.length; i++) {
					editionState.images[i] = batch.images[i];
				}
				editionState.loadedThrough = firstEnd;

				mountFlipbook({
					images: getLoadedImagesSlice(),
					width: editionState.width,
					height: editionState.height,
					isMobile: isMobile,
					totalPages: totalPages
				});

				if (totalPages > firstEnd) {
					setPreloadHint(true);
					loadRemainingPages(pdfUrl, firstEnd + 1, totalPages, token);
				}
			} catch (e) {
				showError('{$load_error}');
			} finally {
				isLoading = false;
			}
		}

		archiveItems.forEach(function (item) {
			item.addEventListener('click', function () {
				var pdfUrl = item.getAttribute('data-edition-pdf');
				var editionKey = item.getAttribute('data-edition-key');
				loadEdition(pdfUrl, editionKey, true);
				if (item.blur) {
					item.blur();
				}
			});
		});

		if (backCurrentBtn) {
			backCurrentBtn.addEventListener('click', function () {
				loadEdition(currentPdfUrl, 'current', true);
				if (backCurrentBtn.blur) {
					backCurrentBtn.blur();
				}
			});
		}

		if (zoomViewport) {
			zoomViewport.addEventListener('pointerdown', function (event) {
				if (zoom <= 1.01) return;
				if (event.button !== undefined && event.button !== 0) return;
				isPanning = true;
				root.classList.add('is-panning');
				panStartX = event.clientX;
				panStartY = event.clientY;
				panOriginX = panX;
				panOriginY = panY;
				if (zoomViewport.setPointerCapture) {
					zoomViewport.setPointerCapture(event.pointerId);
				}
				event.preventDefault();
			});

			zoomViewport.addEventListener('pointermove', function (event) {
				if (!isPanning) return;
				panX = panOriginX + (event.clientX - panStartX);
				panY = panOriginY + (event.clientY - panStartY);
				applyZoomTransform();
			});

			function endPan(event) {
				if (!isPanning) return;
				isPanning = false;
				root.classList.remove('is-panning');
				if (zoomViewport.releasePointerCapture && event && event.pointerId) {
					try { zoomViewport.releasePointerCapture(event.pointerId); } catch (err) {}
				}
			}

			zoomViewport.addEventListener('pointerup', endPan);
			zoomViewport.addEventListener('pointercancel', endPan);
			zoomViewport.addEventListener('lostpointercapture', function () {
				isPanning = false;
				root.classList.remove('is-panning');
			});

			zoomViewport.addEventListener('wheel', function (event) {
				if (!(event.ctrlKey || event.metaKey)) return;
				event.preventDefault();
				var direction = event.deltaY < 0 ? 1 : -1;
				setZoom(zoom + direction * ZOOM_STEP, true);
			}, { passive: false });

			zoomViewport.addEventListener('dblclick', function () {
				if (zoom > 1.01) {
					resetZoom();
				} else {
					setZoom(1.75, false);
				}
			});
		}

		if (mobileQuery.addEventListener) {
			mobileQuery.addEventListener('change', function () {
				var nowMobile = isMagazinMobile();
				root.classList.toggle('tusn-magazin--mobile', nowMobile);
				if (!pageFlip && activePdf) {
					loadEdition(activePdf, root.getAttribute('data-active') || 'current');
				}
				if (pageFlip && typeof pageFlip.update === 'function') {
					pageFlip.update();
				}
			});
		}

		root.addEventListener('keydown', function (event) {
			if (!pageFlip) return;
			if (event.key === 'ArrowLeft') pageFlip.flipPrev();
			if (event.key === 'ArrowRight') tryFlipNext();
			if (event.key === '+' || event.key === '=') setZoom(zoom + ZOOM_STEP, true);
			if (event.key === '-' || event.key === '_') setZoom(zoom - ZOOM_STEP, true);
			if (event.key === '0') resetZoom();
		});

		applyZoomTransform();
		bindControls();
		loadArchiveCovers();
		root.classList.toggle('tusn-magazin--mobile', isMagazinMobile());
		loadEdition(activePdf, root.getAttribute('data-active') || 'current');
	}

	function boot() {
		document.querySelectorAll('.tusn-magazin[data-pdf]').forEach(initMagazin);
	}

	if (document.readyState === 'loading') {
		document.addEventListener('DOMContentLoaded', boot);
	} else {
		boot();
	}
})();
