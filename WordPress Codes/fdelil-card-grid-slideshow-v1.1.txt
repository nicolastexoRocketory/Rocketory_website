function fdelil_cards_normalize_item( $raw ) {
	$tags = array();

	for ( $i = 1; $i <= 3; $i++ ) {
		$label = trim( (string) ( $raw[ "tag_{$i}_label" ] ?? '' ) );
		if ( '' === $label ) {
			continue;
		}
		$tags[] = array(
			'label' => $label,
			'url'   => esc_url_raw( $raw[ "tag_{$i}_url" ] ?? '' ),
			'style' => ( ( $raw[ "tag_{$i}_style" ] ?? 'neutral' ) === 'accent' ) ? 'accent' : 'neutral',
		);
	}

	$maps_url   = trim( (string) ( $raw['maps_url'] ?? '' ) );
	$maps_label = trim( (string) ( $raw['maps_label'] ?? '' ) );
	if ( $maps_url ) {
		$tags[] = array(
			'label' => $maps_label !== '' ? $maps_label : 'GOOGLE MAPS',
			'url'   => esc_url_raw( $maps_url ),
			'style' => 'neutral',
			'icon'  => 'maps',
		);
	}

	$contacts = array();
	for ( $i = 1; $i <= 4; $i++ ) {
		$text = trim( (string) ( $raw[ "contact_{$i}_text" ] ?? '' ) );
		if ( '' === $text ) {
			continue;
		}
		$contacts[] = array(
			'text' => $text,
			'url'  => esc_url_raw( $raw[ "contact_{$i}_url" ] ?? '' ),
			'icon' => sanitize_key( $raw[ "contact_{$i}_icon" ] ?? 'phone' ),
		);
	}

	return array(
		'image'    => esc_url_raw( $raw['image'] ?? '' ),
		'title'    => sanitize_text_field( $raw['title'] ?? '' ),
		'text'     => wp_kses_post( $raw['text'] ?? '' ),
		'alt'      => sanitize_text_field( $raw['alt'] ?? ( $raw['title'] ?? '' ) ),
		'tags'     => $tags,
		'contacts' => $contacts,
	);
}

/* ---------------------------------------------------------
   SHORTCODE
   [fdelil_cards columns="2"]
   [fdelil_card title="..." text="..." tag1="Adresse|URL|accent" maps="https://..." contact1="phone|05741…|tel:…"]
--------------------------------------------------------- */

add_shortcode( 'fdelil_cards', 'fdelil_cards_shortcode' );
add_shortcode( 'fdelil_card', 'fdelil_card_item_shortcode' );

function fdelil_card_parse_pipe( $value ) {
	$parts = array_map( 'trim', explode( '|', (string) $value ) );
	return array(
		0 => $parts[0] ?? '',
		1 => $parts[1] ?? '',
		2 => $parts[2] ?? '',
	);
}

function fdelil_card_item_shortcode( $atts, $content = null ) {
	$atts = shortcode_atts(
		array(
			'image'     => '',
			'title'     => '',
			'text'      => '',
			'alt'       => '',
			'tag1'      => '',
			'tag2'      => '',
			'tag3'      => '',
			'maps'      => '',
			'maps_label'=> 'GOOGLE MAPS',
			'contact1'  => '',
			'contact2'  => '',
			'contact3'  => '',
			'contact4'  => '',
		),
		$atts,
		'fdelil_card'
	);

	$text = $atts['text'] !== '' ? $atts['text'] : trim( (string) $content );
	$raw  = array(
		'image'      => $atts['image'],
		'title'      => $atts['title'],
		'text'       => $text,
		'alt'        => $atts['alt'] ? $atts['alt'] : $atts['title'],
		'maps_url'   => $atts['maps'],
		'maps_label' => $atts['maps_label'],
	);

	foreach ( array( 1, 2, 3 ) as $i ) {
		$p = fdelil_card_parse_pipe( $atts[ "tag{$i}" ] );
		$raw[ "tag_{$i}_label" ] = $p[0];
		$raw[ "tag_{$i}_url" ]   = $p[1];
		$raw[ "tag_{$i}_style" ] = $p[2] ? $p[2] : ( 1 === $i ? 'accent' : 'neutral' );
	}

	foreach ( array( 1, 2, 3, 4 ) as $i ) {
		$p = fdelil_card_parse_pipe( $atts[ "contact{$i}" ] );
		$raw[ "contact_{$i}_icon" ] = $p[0] ? $p[0] : 'phone';
		$raw[ "contact_{$i}_text" ] = $p[1] ? $p[1] : ( $p[0] && false === strpos( $atts[ "contact{$i}" ], '|' ) ? $p[0] : '' );
		// support "text only" or "icon|text|url"
		if ( $p[1] === '' && $p[0] !== '' && ! in_array( $p[0], array( 'phone', 'email', 'whatsapp', 'clock', 'pin', 'user', 'globe' ), true ) ) {
			$raw[ "contact_{$i}_icon" ] = 'phone';
			$raw[ "contact_{$i}_text" ] = $p[0];
			$raw[ "contact_{$i}_url" ]  = $p[1];
		} else {
			$raw[ "contact_{$i}_url" ] = $p[2];
		}
	}

	global $fdelil_cards_buffer;
	if ( ! is_array( $fdelil_cards_buffer ) ) {
		$fdelil_cards_buffer = array();
	}
	$fdelil_cards_buffer[] = fdelil_cards_normalize_item( $raw );

	return '';
}

function fdelil_cards_shortcode( $atts, $content = null ) {
	$atts = shortcode_atts(
		array(
			'columns'    => '4',
			'breakpoint' => '768',
		),
		$atts,
		'fdelil_cards'
	);

	global $fdelil_cards_buffer;
	$fdelil_cards_buffer = array();

	if ( $content ) {
		do_shortcode( $content );
	}

	$items = $fdelil_cards_buffer;
	$fdelil_cards_buffer = array();

	if ( empty( $items ) ) {
		return current_user_can( 'edit_posts' )
			? '<p><em>Card Grid: Bitte Cards im Elementor-Widget hinzufügen oder [fdelil_card] nested verwenden.</em></p>'
			: '';
	}

	return fdelil_cards_render(
		$items,
		array(
			'columns'    => intval( $atts['columns'] ),
			'breakpoint' => intval( $atts['breakpoint'] ),
		)
	);
}

	<script>
	(function () {
		var root = document.getElementById(<?php echo wp_json_encode( $id ); ?>);
		if (!root || root.dataset.fdcReady === '1') return;
		root.dataset.fdcReady = '1';

		var track = root.querySelector('.fdc-track');
		var cards = root.querySelectorAll('.fdc-card');
		var scrollbar = root.querySelector('.fdc-scrollbar');
		var scrollbarTrack = root.querySelector('.fdc-scrollbar__track');
		var scrollbarThumb = root.querySelector('.fdc-scrollbar__thumb');
		var btnPrev = root.querySelector('.fdc-btn--prev');
		var btnNext = root.querySelector('.fdc-btn--next');
		var breakpoint = parseInt(root.getAttribute('data-breakpoint') || '768', 10);
		var count = cards.length || parseInt(root.getAttribute('data-count') || '0', 10);
		var index = 0;
		var scrollTicking = false;
		var scroller = track;
		var dragActive = false;
		var dragMoved = false;
		var dragStartX = 0;
		var dragStartScroll = 0;

		function unlockParentClip() {
			if (!isMobile() || !root) return;
			var el = root.parentElement;
			for (var i = 0; i < 8 && el && el !== document.body; i++) {
				var cs = window.getComputedStyle(el);
				if (cs.overflowX === 'hidden' || cs.overflow === 'hidden' || cs.overflowX === 'clip') {
					el.style.setProperty('overflow-x', 'visible', 'important');
					if (cs.overflow === 'hidden' || cs.overflow === 'clip') {
						el.style.setProperty('overflow-y', cs.overflowY === 'hidden' || cs.overflowY === 'clip' ? 'visible' : cs.overflowY, 'important');
					}
				}
				el = el.parentElement;
			}
		}

		function isMobile() {
			return window.matchMedia('(max-width: ' + breakpoint + 'px)').matches;
		}

		function getGap() {
			if (!track) return 0;
			var styles = window.getComputedStyle(track);
			return parseFloat(styles.columnGap || styles.gap) || 0;
		}

		function getStep() {
			if (!cards.length) return 0;
			return cards[0].getBoundingClientRect().width + getGap();
		}

		function syncScrollbar() {
			if (!scrollbar || !scrollbarTrack || !scrollbarThumb || !scroller) return;

			var scrollWidth = scroller.scrollWidth;
			var clientWidth = scroller.clientWidth;
			var maxScroll = scrollWidth - clientWidth;
			var trackWidth = scrollbarTrack.clientWidth;

			if (maxScroll <= 1 || trackWidth <= 0) {
				scrollbar.classList.add('is-static');
				scrollbar.setAttribute('aria-valuenow', '0');
				return;
			}

			scrollbar.classList.remove('is-static');
			var thumbWidth = Math.max(48, (clientWidth / scrollWidth) * trackWidth);
			var travel = Math.max(0, trackWidth - thumbWidth);
			var ratio = scroller.scrollLeft / maxScroll;
			var thumbLeft = travel * ratio;

			scrollbarThumb.style.width = thumbWidth + 'px';
			scrollbarThumb.style.transform = 'translateX(' + thumbLeft + 'px)';
			scrollbar.setAttribute('aria-valuenow', String(Math.round(ratio * 100)));
		}

		function syncIndexFromScroll() {
			index = indexFromScroll();
			syncScrollbar();
		}

		function indexFromScroll() {
			var step = getStep();
			if (!scroller || step <= 0) return 0;
			return Math.max(0, Math.min(count - 1, Math.round(scroller.scrollLeft / step)));
		}

		function goTo(nextIndex, instant) {
			if (count < 1 || !scroller) return;
			var target = Math.max(0, Math.min(count - 1, nextIndex));
			index = target;

			if (!isMobile()) {
				scroller.scrollLeft = 0;
				syncScrollbar();
				return;
			}

			var left = Math.round(target * getStep());
			if (instant) {
				var prev = scroller.style.scrollBehavior;
				scroller.style.scrollBehavior = 'auto';
				scroller.scrollLeft = left;
				scroller.style.scrollBehavior = prev || '';
			} else if (typeof scroller.scrollTo === 'function') {
				scroller.scrollTo({ left: left, behavior: 'smooth' });
			} else {
				scroller.scrollLeft = left;
			}
			syncScrollbar();
		}

		function scrollFromScrollbar(clientX) {
			if (!scroller || !scrollbarTrack) return;
			var rect = scrollbarTrack.getBoundingClientRect();
			var trackWidth = rect.width;
			var maxScroll = scroller.scrollWidth - scroller.clientWidth;
			if (trackWidth <= 0 || maxScroll <= 0) return;

			var thumbWidth = scrollbarThumb ? scrollbarThumb.offsetWidth : 48;
			var travel = Math.max(0, trackWidth - thumbWidth);
			var clickX = Math.max(0, Math.min(trackWidth, clientX - rect.left));
			var targetLeft = travel > 0 ? ((clickX - thumbWidth / 2) / travel) * maxScroll : 0;

			scroller.scrollLeft = Math.max(0, Math.min(maxScroll, targetLeft));
			syncIndexFromScroll();
		}

		function onScroll() {
			if (!isMobile()) return;
			if (scrollTicking) return;
			scrollTicking = true;
			window.requestAnimationFrame(function () {
				syncIndexFromScroll();
				scrollTicking = false;
			});
		}

		if (btnPrev) {
			btnPrev.addEventListener('click', function () {
				goTo(index - 1);
			});
		}
		if (btnNext) {
			btnNext.addEventListener('click', function () {
				goTo(index + 1);
			});
		}

		if (scrollbar && scrollbarTrack) {
			scrollbarTrack.addEventListener('click', function (e) {
				if (dragActive) return;
				if (dragMoved) {
					dragMoved = false;
					return;
				}
				scrollFromScrollbar(e.clientX);
			});

			scrollbarTrack.addEventListener('pointerdown', function (e) {
				if (!isMobile()) return;
				dragActive = true;
				dragMoved = false;
				dragStartX = e.clientX;
				dragStartScroll = scroller ? scroller.scrollLeft : 0;
				scrollbar.classList.add('is-dragging');
				if (scrollbarTrack.setPointerCapture) {
					scrollbarTrack.setPointerCapture(e.pointerId);
				}
				e.preventDefault();
			});

			scrollbarTrack.addEventListener('pointermove', function (e) {
				if (!dragActive || !scroller || !scrollbarTrack) return;
				var maxScroll = scroller.scrollWidth - scroller.clientWidth;
				var trackWidth = scrollbarTrack.clientWidth;
				var thumbWidth = scrollbarThumb ? scrollbarThumb.offsetWidth : 48;
				var travel = Math.max(0, trackWidth - thumbWidth);
				if (travel <= 0 || maxScroll <= 0) return;

				var deltaX = e.clientX - dragStartX;
				if (Math.abs(deltaX) > 3) dragMoved = true;
				var scrollDelta = (deltaX / travel) * maxScroll;
				scroller.scrollLeft = Math.max(0, Math.min(maxScroll, dragStartScroll + scrollDelta));
				syncIndexFromScroll();
			});

			function endDrag(e) {
				if (!dragActive) return;
				dragActive = false;
				scrollbar.classList.remove('is-dragging');
				if (scrollbarTrack.releasePointerCapture && e && e.pointerId) {
					try { scrollbarTrack.releasePointerCapture(e.pointerId); } catch (err) {}
				}
			}

			scrollbarTrack.addEventListener('pointerup', endDrag);
			scrollbarTrack.addEventListener('pointercancel', endDrag);
			scrollbarTrack.addEventListener('lostpointercapture', function () {
				dragActive = false;
				scrollbar.classList.remove('is-dragging');
			});
		}

		if (scroller) {
			scroller.addEventListener('scroll', onScroll, { passive: true });
		}

		window.addEventListener('resize', function () {
			unlockParentClip();
			goTo(index, true);
			syncScrollbar();
		});

		unlockParentClip();
		goTo(0, true);
		syncScrollbar();
	})();
	</script>
