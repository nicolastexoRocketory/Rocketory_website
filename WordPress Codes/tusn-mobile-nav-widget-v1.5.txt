add_action( 'after_setup_theme', 'tusn_mn_register_menu' );

function tusn_mn_defaults() {

function tusn_mn_get_opts() {
	$saved = get_option( 'tusn_mobile_nav_opts' );
	if ( ! is_array( $saved ) ) {
		$saved = array();
	}
	return wp_parse_args( $saved, tusn_mn_defaults() );
}

/* ============================================================
   3) Walker: Top-Level-Links + Untermenüs als Drill-down
   ============================================================ */

class TUSN_Mobile_Nav_Walker extends Walker_Nav_Menu {
	public $sub_panels = array();

	public function start_el( &$output, $item, $depth = 0, $args = null, $id = 0 ) {
		$title = apply_filters( 'the_title', $item->title, $item->ID );
		$title = apply_filters( 'nav_menu_item_title', $title, $item, $args, $depth );
		$url   = ! empty( $item->url ) ? $item->url : '#';
		$has_children = in_array( 'menu-item-has-children', (array) $item->classes, true );

		if ( 0 === (int) $depth ) {
			if ( $has_children ) {
				$sub_id = 'tusnMnSub-' . (int) $item->ID;
				$output .= '<button type="button" class="tusn-mn__link tusn-mn__drill" data-tusn-mn-sub="' . esc_attr( $sub_id ) . '" aria-expanded="false" aria-controls="' . esc_attr( $sub_id ) . '">';
				$output .= esc_html( $title );
				$output .= ' <span aria-hidden="true">→</span></button>';
				if ( ! isset( $this->sub_panels[ $sub_id ] ) ) {
					$this->sub_panels[ $sub_id ] = array(
						'title'  => $title,
						'parent' => $item,
						'items'  => array(),
					);
				} else {
					$this->sub_panels[ $sub_id ]['title']  = $title;
					$this->sub_panels[ $sub_id ]['parent'] = $item;
				}
			} else {
				$output .= '<a class="tusn-mn__link" href="' . esc_url( $url ) . '" data-tusn-mn-close>';
				$output .= esc_html( $title );
				$output .= '</a>';
			}
		}
	}

	public function display_element( $element, &$children_elements, $max_depth, $depth, $args, &$output ) {
		if ( ! $element ) {
			return;
		}
		$id_field = $this->db_fields['id'];
		$id       = $element->$id_field;

		// Untermenü-Einträge für Drill-Panel sammeln
		if ( 0 === (int) $depth && ! empty( $children_elements[ $id ] ) ) {
			$sub_id = 'tusnMnSub-' . (int) $id;
			if ( ! isset( $this->sub_panels[ $sub_id ] ) ) {
				$this->sub_panels[ $sub_id ] = array(
					'title'  => $element->title,
					'parent' => $element,
					'items'  => array(),
				);
			}
			foreach ( $children_elements[ $id ] as $child ) {
				$this->sub_panels[ $sub_id ]['items'][] = $child;
			}
		}

		parent::display_element( $element, $children_elements, $max_depth, $depth, $args, $output );
	}

	public function start_lvl( &$output, $depth = 0, $args = null ) {
		// Untermenüs werden als eigene Panels gerendert, nicht als nested UL.
	}

	public function end_lvl( &$output, $depth = 0, $args = null ) {}

	public function end_el( &$output, $item, $depth = 0, $args = null ) {}
}

	(function () {
	  if (window.__tusnMnBound) return;
	  window.__tusnMnBound = true;

	  function ready(fn) {
	    if (document.readyState !== 'loading') fn();
	    else document.addEventListener('DOMContentLoaded', fn);
	  }

	  ready(function () {
	    var root = document.getElementById('tusnMobileNav');
	    if (!root) return;
	    var btn = document.getElementById('tusnMnToggle');
	    var panel = document.getElementById('tusnMnPanel');
	    var closeBtn = document.getElementById('tusnMnClose');
	    var reduce = window.matchMedia('(prefers-reduced-motion: reduce)');

	    function resetSubs(skipFocus) {
	      var active = panel.querySelector('.tusn-mn__view--sub.is-active');
	      var trigger = active ? panel.querySelector('[data-tusn-mn-sub="' + active.id + '"]') : null;
	      panel.classList.remove('is-sub-open');
	      var subnav = panel.querySelector('.tusn-mn__subnav');
	      if (subnav) subnav.setAttribute('aria-hidden', 'true');
	      panel.querySelectorAll('.tusn-mn__view--sub').forEach(function (v) {
	        v.classList.remove('is-active');
	        v.setAttribute('aria-hidden', 'true');
	      });
	      panel.querySelectorAll('[data-tusn-mn-sub]').forEach(function (b) {
	        b.setAttribute('aria-expanded', 'false');
	      });
	      if (!skipFocus && trigger) trigger.focus({ preventScroll: true });
	    }

	    function closeMenu() {
	      resetSubs(true);
	      panel.classList.remove('is-open', 'is-reveal');
	      panel.setAttribute('inert', '');
	      btn.classList.remove('is-open');
	      btn.setAttribute('aria-expanded', 'false');
	      btn.setAttribute('aria-label', 'Menü öffnen');
	      document.body.style.overflow = '';
	      btn.focus({ preventScroll: true });
	    }

	    function openMenu() {
	      panel.classList.add('is-open');
	      panel.removeAttribute('inert');
	      btn.classList.add('is-open');
	      btn.setAttribute('aria-expanded', 'true');
	      btn.setAttribute('aria-label', 'Menü schließen');
	      document.body.style.overflow = 'hidden';
	      if (!reduce.matches) {
	        panel.classList.remove('is-reveal');
	        requestAnimationFrame(function () {
	          requestAnimationFrame(function () { panel.classList.add('is-reveal'); });
	        });
	      } else {
	        panel.classList.add('is-reveal');
	      }
	    }

	    btn.addEventListener('click', function () {
	      panel.classList.contains('is-open') ? closeMenu() : openMenu();
	    });
	    if (closeBtn) closeBtn.addEventListener('click', closeMenu);

	    panel.querySelectorAll('[data-tusn-mn-close]').forEach(function (el) {
	      el.addEventListener('click', closeMenu);
	    });

	    panel.querySelectorAll('[data-tusn-mn-sub]').forEach(function (drill) {
	      drill.addEventListener('click', function () {
	        var id = drill.getAttribute('data-tusn-mn-sub');
	        var sub = document.getElementById(id);
	        if (!sub) return;
	        resetSubs(true);
	        sub.classList.add('is-active');
	        sub.setAttribute('aria-hidden', 'false');
	        panel.classList.add('is-sub-open');
	        var subnav = panel.querySelector('.tusn-mn__subnav');
	        if (subnav) subnav.setAttribute('aria-hidden', 'false');
	        drill.setAttribute('aria-expanded', 'true');
	      });
	    });

	    panel.querySelectorAll('[data-tusn-mn-back]').forEach(function (b) {
	      b.addEventListener('click', function () { resetSubs(false); });
	    });

	    document.addEventListener('keydown', function (e) {
	      if (e.key === 'Escape' && panel.classList.contains('is-open')) {
	        if (panel.classList.contains('is-sub-open')) resetSubs(false);
	        else closeMenu();
	      }
	    });
	  });
	})();

add_action( 'wp_body_open', 'tusn_mn_auto_print', 5 );
add_action( 'wp_footer', 'tusn_mn_auto_print_fallback', 5 );

function tusn_mn_auto_print() {
	if ( is_admin() ) {
		return;
	}
	echo tusn_mn_render(); // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
}

function tusn_mn_auto_print_fallback() {
	// Themes ohne wp_body_open: einmal im Footer ausgeben (fixed bleibt am Viewport).
	if ( ! did_action( 'wp_body_open' ) ) {
		echo tusn_mn_render(); // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
	}
}
